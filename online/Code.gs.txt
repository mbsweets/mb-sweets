/**
 * MB Sweets — ऑनलाइन ऑर्डर रजिस्टर · कोड संस्करण 5 (6 अक्टूबर 2026)
 * Google Sheet + Apps Script. इस कोड में कुछ बदलने की ज़रूरत नहीं है।
 *
 * ग्राहक का मेन्यू यहाँ ऑर्डर भेजता है, ऑर्डर "Orders" शीट में लिखा जाता है,
 * आपके फोन पर बिलिंग ऐप में सीधी घंटी (web push) जाती है — घंटी चालू न हो तो ntfy से —
 * और बिलिंग ऐप यहीं से ऑर्डर पढ़ता है।
 * दुकान की गुप्त चाबी (key) और ntfy का नाम पहली बार सेटअप में अपने-आप बनते हैं।
 *
 * दूध-दही रिमाइंडर: बिलिंग ऐप आने वाले ऑर्डर का हिसाब "Plan" शीट में रखता है।
 * रोज़ दोपहर करीब 2 से 2:30 के बीच फोन पर घंटी — परसों की मिठाई का दूध, कल का दही/दूध/पनीर।
 *
 * ⭐ माँ भगवती मिठास पॉइंट्स: असली हिसाब मालिक के फ़ोन (बिलिंग ऐप) में है। ऐप हर सदस्य का छोटा सार
 * "Points" शीट में भेजता है ताकि ग्राहक अपने पॉइंट ख़ुद देख सके — WhatsApp वाले लिंक से पूरा हिसाब,
 * सिर्फ़ मोबाइल नंबर से सिर्फ़ पॉइंट (नाम और ख़रीद नहीं)।
 */
var VERSION = 5;

var SHEET_NAME = 'Orders';
var BILLING_URL = 'https://mbsweets.github.io/mb-sweets/';
var HEAD = ['code', 'created', 'status', 'name', 'phone', 'mode', 'village', 'address', 'landmark',
  'lat', 'lng', 'distKm', 'when', 'slot', 'summary', 'total', 'items', 'custom', 'cakeMsg', 'note',
  'lang', 'updated', 'statusNote', 'confirmedTotal'];
var SLOTS = ['सुबह 8–11', 'दोपहर 11–2', 'दोपहर 2–5', 'शाम 5–8'];
var STATUSES = ['new', 'accepted', 'out', 'delivered', 'rejected'];

/* सेटअप: कोड चिपकाने के बाद ऊपर ▶ Run दबाएं — यही फ़ंक्शन चलेगा
   (शीट जोड़ता है, दूध-दही रिमाइंडर का रोज़ का टाइमर लगाता है और Google से अनुमति लेता है)।
   दोबारा चलाने से कुछ खराब नहीं होता — ऑर्डर और चाबी वैसे ही रहते हैं। */
function setupSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  PropertiesService.getScriptProperties().setProperty('SHEET_ID', ss.getId());
  sheet_();
  setupReminder();
  try {
    UrlFetchApp.fetch('https://ntfy.sh/v1/health', { method: 'get', muteHttpExceptions: true });
  } catch (e) {
    Logger.log('⚠️ ntfy अभी नहीं मिला (इंटरनेट की दिक्कत) — बाकी सेटअप हो गया, Deploy के बाद टेस्ट नोटिफिकेशन भेजकर देखें');
  }
  Logger.log('✅ सेटअप ठीक है (रिमाइंडर भी चालू) — अब Deploy करें');
}

/* रोज़ का टाइमर: दोपहर ~2 बजे और ~2:40 बजे (दूसरा सिर्फ तब चलता है जब पहला न पहुँचा हो) */
function setupReminder() {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === 'dailyReminder') ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('dailyReminder').timeBased().atHour(14).nearMinute(5).everyDays(1).inTimezone('Asia/Kolkata').create();
  ScriptApp.newTrigger('dailyReminder').timeBased().atHour(14).nearMinute(40).everyDays(1).inTimezone('Asia/Kolkata').create();
  props_().setProperty('REMINDER', 'on');
  planSheet_();
  Logger.log('✅ दूध-दही रिमाइंडर चालू — रोज़ दोपहर करीब 2 से 2:30 के बीच');
}

/* रिमाइंडर बंद करना हो तो इसे ▶ Run करें */
function stopReminder() {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === 'dailyReminder') ScriptApp.deleteTrigger(t);
  });
  props_().setProperty('REMINDER', 'off');
  Logger.log('रिमाइंडर बंद');
}

/* ---------- helpers ---------- */
function props_() { return PropertiesService.getScriptProperties(); }
function out_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
function ss_() {
  var ss = null;
  try { ss = SpreadsheetApp.getActiveSpreadsheet(); } catch (e) { ss = null; }
  if (!ss) {
    var id = props_().getProperty('SHEET_ID');
    if (id) ss = SpreadsheetApp.openById(id);
  }
  return ss;
}
function sheet_() {
  var ss = ss_();
  var sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) {
    sh = ss.insertSheet(SHEET_NAME);
    sh.appendRow(HEAD);
    sh.setFrozenRows(1);
  }
  return sh;
}
function str_(v, max) { v = (v === null || v === undefined) ? '' : String(v); v = v.replace(/[\u0000-\u0008\u000B-\u001F]/g, ''); return v.length > max ? v.slice(0, max) : v; }
function num_(v) { var n = parseFloat(v); return isFinite(n) ? n : 0; }
function rnd_(n) {
  var abc = 'abcdefghijkmnpqrstuvwxyz23456789', s = '';
  for (var i = 0; i < n; i++) s += abc.charAt(Math.floor(Math.random() * abc.length));
  return s;
}
function keyOk_(k) { var key = props_().getProperty('KEY'); return !!key && String(k || '') === key; }
function rowToObj_(r) {
  var o = {};
  for (var i = 0; i < HEAD.length; i++) o[HEAD[i]] = r[i];
  try { o.items = JSON.parse(o.items || '[]'); } catch (e) { o.items = []; }
  try { o.custom = JSON.parse(o.custom || '[]'); } catch (e) { o.custom = []; }
  o.created = o.created instanceof Date ? o.created.toISOString() : String(o.created || '');
  o.updated = o.updated instanceof Date ? o.updated.toISOString() : String(o.updated || '');
  o.phone = String(o.phone || '').replace(/^'/, '');
  o.when = String(o.when || '').replace(/^'/, '');
  o.code = String(o.code || '');
  return o;
}
function findRow_(sh, code) {
  var last = sh.getLastRow();
  if (last < 2) return -1;
  var codes = sh.getRange(2, 1, last - 1, 1).getValues();
  for (var i = codes.length - 1; i >= 0; i--) if (String(codes[i][0]) === code) return i + 2;
  return -1;
}
function rate_(bucket, limit, seconds) {
  var c = CacheService.getScriptCache();
  var n = parseInt(c.get(bucket) || '0', 10);
  if (n >= limit) return false;
  c.put(bucket, String(n + 1), seconds);
  return true;
}
function rupee_(n) { return '₹' + Math.round(num_(n)); }
function whenText_(w, s, mode) {
  if (w === 'asap') return mode === 'p' ? 'जल्दी (30 मिनट में तैयार)' : 'जल्दी (30–60 मिनट)';
  var d = String(w).split('-');
  var months = ['जनवरी', 'फरवरी', 'मार्च', 'अप्रैल', 'मई', 'जून', 'जुलाई', 'अगस्त', 'सितंबर', 'अक्टूबर', 'नवंबर', 'दिसंबर'];
  var t = d.length === 3 ? (parseInt(d[2], 10) + ' ' + months[parseInt(d[1], 10) - 1]) : w;
  return t + (s >= 0 && s < SLOTS.length ? ', ' + SLOTS[s] : '');
}

/* ---------- notification to the owner's phone ----------
   पहले बिलिंग ऐप की अपनी घंटी (web push); वो चालू न हो या न पहुँचे तो ntfy ऐप। */
function notify_(title, message, priority, click, tags, tag) {
  return deliver_(title, message, priority, click, tags, tag).ok;
}
var LAST_DELIVERY_ = null;
function deliver_(title, message, priority, click, tags, tag) {
  var data = { t: str_(title, 200), b: str_(message, 1200), u: click || (BILLING_URL + '#online'), g: tag || '',
    s: (priority || 4) < 3, i: Date.now() };
  var res = { ok: false, via: '', detail: [] };
  try { inboxAdd_(data); } catch (e) { }
  var subs = subs_();
  if (subs.length) {
    var keep = [];
    subs.forEach(function (sub) {
      var r = pushOne_(sub, data, priority);
      res.detail.push(r.mode + ':' + r.code + (r.err ? ' ' + r.err : ''));
      if (r.code >= 200 && r.code < 300) res.ok = true;
      if (r.code !== 404 && r.code !== 410) keep.push(sub);
    });
    if (keep.length !== subs.length) saveSubs_(keep);
    if (res.ok) { res.via = 'push'; LAST_DELIVERY_ = res; return res; }
  }
  var n = ntfy_(title, message, priority, click, tags);
  res.detail.push('ntfy:' + n);
  if (n === 'ok') { res.ok = true; res.via = 'ntfy'; }
  LAST_DELIVERY_ = res;
  return res;
}
function ntfy_(title, message, priority, click, tags) {
  var topic = props_().getProperty('TOPIC');
  if (!topic) return 'off';
  try {
    var res = UrlFetchApp.fetch('https://ntfy.sh/', {
      method: 'post', contentType: 'application/json', muteHttpExceptions: true,
      payload: JSON.stringify({ topic: topic, title: title, message: message, priority: priority || 4,
        tags: tags || ['shopping_bags'], click: click || (BILLING_URL + '#online') })
    });
    var c = res.getResponseCode();
    return c < 300 ? 'ok' : String(c);
  } catch (e) { return String(e && e.message || e).slice(0, 80); }
}

/* ---------- web app entry points ---------- */
function doGet(e) {
  var p = (e && e.parameter) || {};
  try {
    if (p.action === 'ping') return out_({ ok: true, ready: !!props_().getProperty('KEY'), v: VERSION, reminder: props_().getProperty('REMINDER') === 'on', vapid: wpVapid_().pub });
    if (p.action === 'inbox') {
      if (!keyOk_(p.key)) return out_({ ok: false, error: 'key' });
      return out_({ ok: true, items: inbox_() });
    }
    if (p.action === 'init') return out_(init_());
    if (p.action === 'status') return out_(status_(p.code, p.p4));
    if (p.action === 'list') return out_(list_(p.key, p.days));
    if (p.action === 'lpview') return out_(lpView_(p.t));
    if (p.action === 'lppts') return out_(lpPts_(p.p));
    if (p.action === 'testnotify') {
      if (!keyOk_(p.key)) return out_({ ok: false, error: 'key' });
      return out_(deliver_('✅ MB Sweets — टेस्ट', 'घंटी ठीक से आ रही है। नया ऑनलाइन ऑर्डर आने पर ऐसे ही बजेगी।', 4, BILLING_URL + '#online', null, 'test'));
    }
    if (p.action === 'testplan') {
      if (!keyOk_(p.key)) return out_({ ok: false, error: 'key' });
      return out_(sendReminder_(true));
    }
    return out_({ ok: false, error: 'action' });
  } catch (err) {
    return out_({ ok: false, error: 'server', detail: String(err) });
  }
}

function doPost(e) {
  var body = {};
  try { body = JSON.parse((e && e.postData && e.postData.contents) || '{}'); } catch (err) { return out_({ ok: false, error: 'json' }); }
  try {
    if (body.action === 'create') return out_(create_(body.order || {}));
    if (body.action === 'update') return out_(update_(body));
    if (body.action === 'list') return out_(list_(body.key, body.days));
    if (body.action === 'plan') return out_(plan_(body));
    if (body.action === 'pushsub') return out_(pushSub_(body));
    if (body.action === 'pushunsub') return out_(pushUnsub_(body));
    if (body.action === 'lpsync') return out_(lpSync_(body));
    return out_({ ok: false, error: 'action' });
  } catch (err) {
    return out_({ ok: false, error: 'server', detail: String(err) });
  }
}

/* ---------- first-time setup: makes the secret key and the ntfy name ---------- */
function init_() {
  var pr = props_();
  if (pr.getProperty('KEY')) return { ok: false, error: 'already' };
  var ss = ss_();
  if (!ss) return { ok: false, error: 'nosheet' };
  pr.setProperty('SHEET_ID', ss.getId());
  var key = rnd_(24), topic = 'mbsweets-' + rnd_(12);
  pr.setProperty('KEY', key);
  pr.setProperty('TOPIC', topic);
  sheet_();
  return { ok: true, key: key, topic: topic };
}

/* run this once from the editor (▶ Run) only if you ever need a fresh key */
function resetSetup() {
  var pr = props_();
  pr.deleteProperty('KEY');
  pr.deleteProperty('TOPIC');
}

/* ---------- customer places an order ---------- */
function create_(o) {
  if (o.hp) return { ok: true, code: 'X' };                       // bots fill the hidden field
  var phone = str_(o.p, 15).replace(/\D/g, '');
  if (!/^[6-9]\d{9}$/.test(phone)) return { ok: false, error: 'phone' };
  var name = str_(o.n, 60).trim();
  if (name.length < 2) return { ok: false, error: 'name' };
  var mode = o.m === 'p' ? 'p' : 'd';
  var items = Array.isArray(o.i) ? o.i.slice(0, 60).filter(function (r) {
    return Array.isArray(r) && r.length >= 2 && /^[a-z0-9]{1,12}$/i.test(String(r[0])) && num_(r[1]) > 0 && num_(r[1]) <= 500;
  }).map(function (r) { return [String(r[0]), num_(r[1])]; }) : [];
  var custom = Array.isArray(o.x) ? o.x.slice(0, 20).filter(function (r) {
    return Array.isArray(r) && r.length >= 2 && str_(r[1], 300).trim();
  }).map(function (r) { return [r[0] === 'cake' ? 'cake' : 'pack', str_(r[1], 300)]; }) : [];
  if (!items.length && !custom.length) return { ok: false, error: 'empty' };
  var when = (o.w === 'asap' || /^\d{4}-\d{2}-\d{2}$/.test(String(o.w))) ? String(o.w) : 'asap';
  var slot = (isFinite(o.s) && o.s >= 0 && o.s < SLOTS.length) ? Number(o.s) : -1;
  var code = /^W\d{4}-\d{5}$/.test(String(o.c)) ? String(o.c) : ('W' + Utilities.formatDate(new Date(), 'Asia/Kolkata', 'ddMM-HHmm') + Math.floor(Math.random() * 10));

  if (!rate_('rl_p_' + phone, 6, 600)) return { ok: false, error: 'rate' };
  if (!rate_('rl_all', 120, 3600)) return { ok: false, error: 'rate' };

  var lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    var sh = sheet_();
    if (findRow_(sh, code) > 0) return { ok: true, code: code, dup: true };
    var lat = '', lng = '';
    if (Array.isArray(o.l) && o.l.length === 2 && isFinite(o.l[0]) && isFinite(o.l[1])) { lat = Number(o.l[0]); lng = Number(o.l[1]); }
    var now = new Date();
    sh.appendRow([code, now, 'new', name, "'" + phone, mode, str_(o.g, 80), str_(o.a, 200), str_(o.k, 120),
      lat, lng, isFinite(o.dk) ? Number(o.dk) : '', "'" + when, slot, str_(o.sm, 500), num_(o.t),
      JSON.stringify(items), JSON.stringify(custom), str_(o.cm, 120), str_(o.nt, 300),
      o.lang === 'en' ? 'en' : 'hi', now, '', '']);
  } finally {
    lock.releaseLock();
  }
  var n = items.length + custom.length;
  notify_('🛍️ नया ऑर्डर ' + rupee_(o.t) + (custom.length ? ' +' : ''),
    whenText_(when, slot, mode) + ' · ' + (mode === 'p' ? '🏪 दुकान से लेंगे' : '🚚 डिलीवरी') + ' · ' + n + ' आइटम\nदबाकर बिलिंग ऐप में देखें', 5, null, null, 'order-' + code);
  return { ok: true, code: code };
}

/* ---------- customer checks their order status ---------- */
function status_(code, p4) {
  code = str_(code, 20);
  var sh = sheet_();
  var row = findRow_(sh, code);
  if (row < 0) return { ok: false, error: 'notfound' };
  var o = rowToObj_(sh.getRange(row, 1, 1, HEAD.length).getValues()[0]);
  if (String(o.phone).slice(-4) !== String(p4 || '')) return { ok: false, error: 'notfound' };
  return { ok: true, code: o.code, status: o.status, statusNote: o.statusNote, total: o.total,
    confirmedTotal: o.confirmedTotal, mode: o.mode, when: o.when, slot: o.slot, updated: o.updated };
}

/* ---------- billing app reads the orders ---------- */
function list_(key, days) {
  if (!keyOk_(key)) return { ok: false, error: 'key' };
  var sh = sheet_();
  var last = sh.getLastRow();
  if (last < 2) return { ok: true, orders: [], now: new Date().toISOString() };
  var from = Math.max(2, last - 499);
  var rows = sh.getRange(from, 1, last - from + 1, HEAD.length).getValues();
  var since = Date.now() - Math.min(60, Math.max(1, num_(days) || 10)) * 86400000;
  var list = [];
  rows.forEach(function (r) {
    var o = rowToObj_(r);
    var t = new Date(o.created).getTime();
    if (o.status === 'new' || o.status === 'accepted' || o.status === 'out' || t >= since) list.push(o);
  });
  return { ok: true, orders: list, now: new Date().toISOString() };
}

/* ---------- billing app changes a status ---------- */
function update_(b) {
  if (!keyOk_(b.key)) return { ok: false, error: 'key' };
  var status = String(b.status || '');
  if (STATUSES.indexOf(status) < 0) return { ok: false, error: 'status' };
  var lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    var sh = sheet_();
    var row = findRow_(sh, str_(b.code, 20));
    if (row < 0) return { ok: false, error: 'notfound' };
    sh.getRange(row, HEAD.indexOf('status') + 1).setValue(status);
    sh.getRange(row, HEAD.indexOf('updated') + 1).setValue(new Date());
    if (b.note !== undefined) sh.getRange(row, HEAD.indexOf('statusNote') + 1).setValue(str_(b.note, 200));
    if (b.total !== undefined && b.total !== null && b.total !== '') sh.getRange(row, HEAD.indexOf('confirmedTotal') + 1).setValue(num_(b.total));
  } finally {
    lock.releaseLock();
  }
  return { ok: true };
}

/* ---------- ⭐ माँ भगवती मिठास पॉइंट्स — ग्राहक ख़ुद देखे ----------
   बिलिंग ऐप हर सदस्य का सार भेजता है (id, फ़ोन, लिंक की चाबी, हिसाब)। यहाँ कुछ जोड़ा-घटाया नहीं जाता —
   पॉइंट सिर्फ़ मालिक के फ़ोन से चढ़ते हैं। */
var LP_SHEET = 'Points';
var LP_HEAD = ['id', 'phone', 'token', 'data', 'updated'];
function lpSheet_() {
  var ss = ss_();
  var sh = ss.getSheetByName(LP_SHEET);
  if (!sh) {
    sh = ss.insertSheet(LP_SHEET);
    sh.appendRow(LP_HEAD);
    sh.setFrozenRows(1);
  }
  return sh;
}
function lpClean_(v) { return String(v === null || v === undefined ? '' : v).replace(/^'/, ''); }
function lpSync_(b) {
  if (!keyOk_(b.key)) return { ok: false, error: 'key' };
  if (!Array.isArray(b.list)) return { ok: false, error: 'list' };
  var list = b.list.slice(0, 60), done = [];
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var sh = lpSheet_();
    var last = sh.getLastRow(), idx = {};
    if (last >= 2) sh.getRange(2, 1, last - 1, 1).getValues().forEach(function (r, i) { idx[lpClean_(r[0])] = i + 2; });
    var dels = [], now = new Date();
    list.forEach(function (x) {
      if (!x || typeof x !== 'object') return;
      var id = str_(x.id, 40);
      if (!/^[A-Za-z0-9_-]{1,40}$/.test(id)) return;
      if (x.del) {
        if (idx[id]) { dels.push(idx[id]); delete idx[id]; }
        done.push(id);
        return;
      }
      var phone = str_(x.p, 15).replace(/\D/g, '');
      var tok = str_(x.t, 40);
      if (!/^[6-9]\d{9}$/.test(phone) || !/^[A-Za-z0-9]{16,40}$/.test(tok)) return;
      var data = JSON.stringify(x.d && typeof x.d === 'object' ? x.d : {});
      if (data.length > 20000) return;
      var row = [id, "'" + phone, "'" + tok, data, now];
      if (idx[id]) sh.getRange(idx[id], 1, 1, LP_HEAD.length).setValues([row]);
      else { sh.appendRow(row); idx[id] = sh.getLastRow(); }
      done.push(id);
    });
    dels.sort(function (a, c) { return c - a; }).forEach(function (r) { sh.deleteRow(r); });
  } finally {
    lock.releaseLock();
  }
  return { ok: true, done: done };
}
function lpFind_(col, val) {
  var sh = ss_().getSheetByName(LP_SHEET);
  if (!sh || sh.getLastRow() < 2) return null;
  var n = sh.getLastRow() - 1;
  var vals = sh.getRange(2, col, n, 1).getValues();
  for (var i = n - 1; i >= 0; i--) {
    if (lpClean_(vals[i][0]) !== val) continue;
    try { return JSON.parse(String(sh.getRange(i + 2, 4).getValue() || '')); } catch (e) { return null; }
  }
  return null;
}

function wpUtf8_(s) {
  var out = [];
  for (var i = 0; i < s.length; i++) {
    var c = s.charCodeAt(i);
    if (c >= 0xd800 && c <= 0xdbff && i + 1 < s.length) {
      var d = s.charCodeAt(i + 1);
      if (d >= 0xdc00 && d <= 0xdfff) { c = 0x10000 + ((c - 0xd800) << 10) + (d - 0xdc00); i++; }
    }
    if (c < 0x80) out.push(c);
    else if (c < 0x800) out.push(0xc0 | (c >> 6), 0x80 | (c & 63));
    else if (c < 0x10000) out.push(0xe0 | (c >> 12), 0x80 | ((c >> 6) & 63), 0x80 | (c & 63));
    else out.push(0xf0 | (c >> 18), 0x80 | ((c >> 12) & 63), 0x80 | ((c >> 6) & 63), 0x80 | (c & 63));
  }
  return out;
}

function wpMul_(a, b, M) {

  for (i = 0; i < 18; i++) t[i] = 0;

    C = 0;
    for (j = 0; j < 16; j++) { s = t[j] + a[j] * bi + C; lo = s & 0xffff; t[j] = lo; C = (s - lo) * 0.0000152587890625; }
    s = t[16] + C; lo = s & 0xffff; t[16] = lo; t[17] = (s - lo) * 0.0000152587890625;
